+from fastapi import APIRouter, Depends, HTTPException
+from sqlalchemy.orm import Session
+from typing import List
+from app.models.image import Image
+from app.schemas.image import ImageCreate, ImageUpdate, ImageResponse
+from app.dependencies import get_db
+
+router = APIRouter()
+
+@router.get("/", response_model=List[ImageResponse])
+def get_images(skip: int = 0, limit: int = 100, db: Session = Depends(get_db)):
+    return db.query(Image).offset(skip).limit(limit).all()
+
+@router.post("/", response_model=ImageResponse)
+def create_image(image: ImageCreate, db: Session = Depends(get_db)):
+    db_image = Image(**image.model_dump())
+    db.add(db_image)
+    db.commit()
+    db.refresh(db_image)
+    return db_image
+
+@router.get("/{image_id}", response_model=ImageResponse)
+def get_image(image_id: int, db: Session = Depends(get_db)):
+    image = db.query(Image).filter(Image.id == image_id).first()
+    if image is None:
+        raise HTTPException(status_code=404, detail="Image not found")
+    return image
+
+@router.put("/{image_id}", response_model=ImageResponse)
+def update_image(image_id: int, image_update: ImageUpdate, db: Session = Depends(get_db)):
+    db_image = db.query(Image).filter(Image.id == image_id).first()
+    if db_image is None:
+        raise HTTPException(status_code=404, detail="Image not found")
+    
+    update_data = image_update.model_dump(exclude_unset=True)
+    for key, value in update_data.items():
+        setattr(db_image, key, value)
+        
+    db.commit()
+    db.refresh(db_image)
+    return db_image
+
+@router.delete("/{image_id}")
+def delete_image(image_id: int, db: Session = Depends(get_db)):
+    db_image = db.query(Image).filter(Image.id == image_id).first()
+    if db_image is None:
+        raise HTTPException(status_code=404, detail="Image not found")
+        
+    db.delete(db_image)
+    db.commit()
+    return {"ok": True}

+import os
+import requests
+
+BASE_URL = os.getenv("BACKEND_URL", "http://127.0.0.1:8000/api/v1")
+
+def test_prompt_crud():
+    # Create
+    create_resp = requests.post(f"{BASE_URL}/prompts/", json={"content": "Test prompt content"})
+    assert create_resp.status_code == 200
+    prompt_data = create_resp.json()
+    assert "id" in prompt_data
+    assert prompt_data["content"] == "Test prompt content"
+    prompt_id = prompt_data["id"]
+
+    # Read All
+    list_resp = requests.get(f"{BASE_URL}/prompts/")
+    assert list_resp.status_code == 200
+    assert len(list_resp.json()) > 0
+
+    # Read One
+    get_resp = requests.get(f"{BASE_URL}/prompts/{prompt_id}")
+    assert get_resp.status_code == 200
+    assert get_resp.json()["id"] == prompt_id
+
+    # Update
+    update_resp = requests.put(f"{BASE_URL}/prompts/{prompt_id}", json={"content": "Updated prompt content"})
+    assert update_resp.status_code == 200
+    assert update_resp.json()["content"] == "Updated prompt content"
+
+    # Delete
+    delete_resp = requests.delete(f"{BASE_URL}/prompts/{prompt_id}")
+    assert delete_resp.status_code == 200
+
+    # Verify Delete
+    get_deleted = requests.get(f"{BASE_URL}/prompts/{prompt_id}")
+    assert get_deleted.status_code == 404
+
+def test_wildcard_crud():
+    # Create
+    create_resp = requests.post(f"{BASE_URL}/wildcards/", json={"filename": "test.txt", "content": "test,wildcard"})
+    assert create_resp.status_code == 200
+    wildcard_data = create_resp.json()
+    assert wildcard_data["filename"] == "test.txt"
+    wildcard_id = wildcard_data["id"]
+
+    # Update
+    update_resp = requests.put(f"{BASE_URL}/wildcards/{wildcard_id}", json={"filename": "test2.txt", "content": "updated"})
+    assert update_resp.status_code == 200
+    assert update_resp.json()["filename"] == "test2.txt"
+
+    # Delete
+    requests.delete(f"{BASE_URL}/wildcards/{wildcard_id}")
+
